Zanim cokolwiek dotkniesz

Wszystkie szacunki CPU w tym dokumencie są szacunkami. Renderer jest programowy i pisze do Uint32Array, więc każda zmiana w wewnętrznej pętli jest odczuwalna, a różnice między maszynami są duże. Dodaj licznik przed pierwszą zmianą — inaczej nie będziesz wiedział, który krok kosztował.

/* ---------- szum proceduralny (kafelkujący) ---------- */
function hash2(x,y,seed){
  let h = Math.imul(x,374761393) + Math.imul(y,668265263) + Math.imul(seed,1274126177);
  h = Math.imul(h ^ (h>>>13), 1274126177);
  return ((h ^ (h>>>16))>>>0) / 4294967296;
}
const smooth = t => t*t*(3-2*t);

// u,v w [0,1). `per` to liczba komórek siatki na całą teksturę —
// krata zawija się mod `per`, więc tekstura kafelkuje się bez szwu.
function vnoise(u,v,per,seed){
  const x=u*per, y=v*per;
  const xi=Math.floor(x), yi=Math.floor(y);
  const xf=smooth(x-xi), yf=smooth(y-yi);
  const x0=((xi%per)+per)%per, y0=((yi%per)+per)%per;
  const x1=(x0+1)%per,    y1=(y0+1)%per;
  return lerp(lerp(hash2(x0,y0,seed), hash2(x1,y0,seed), xf),
              lerp(hash2(x0,y1,seed), hash2(x1,y1,seed), xf), yf);
}

// fBm: każda kolejna oktawa dwa razy gęstsza i dwa razy cichsza
function fbm(u,v,base,seed,oct){
  let s=0, amp=0.5, norm=0, f=base;
  for(let o=0;o<oct;o++){
    s += vnoise(u,v,f,seed+o*131)*amp;
    norm += amp; amp *= 0.5; f *= 2;
  }
  return s/norm;
}

Pułapki
Usuń ciemne pasy g.fillRect(0,0,64,2) i fillRect(0,62,64,2) z końca texBark. Dawniej maskowały szew kafla; przy szumie kafelkującym szwu nie ma, a pasy stają się widocznymi poziomymi liniami na każdym pniu.

Koszt
Kod +2 KB. RAM bez zmian. CPU w runtime zero — to koszt generacji. Przy 64² i pięciu teksturach ok. 6 ms na biom; przy 128² ok. 25 ms, raz na poziom, w rebuildArt.
Sprawdzian
Podejdź wplotem do ściany z kory. Powinieneś widzieć pionowe bruzdy i sęki, a nie poziome paski. Odsuń się i obróć: brak pulsowania jasności na granicach kafli.

Pakiet AMieszanka

1/2 — budowa piramidy, obok shadeLevels
const MIP_N = 3;

// filtr pudełkowy 2×2; teksele w pełni przezroczyste nie brudzą średniej,
// bo inaczej sprite'y dostają aureolę w kolorze tła
function downsample(src,w,h){
  const w2=w>>1, h2=h>>1, out=new Uint32Array(w2*h2);
  for(let y=0;y<h2;y++) for(let x=0;x<w2;x++){
    let r=0,g=0,b=0,n=0;
    for(let dy=0;dy<2;dy++) for(let dx=0;dx<2;dx++){
      const c=src[(y*2+dy)*w + x*2+dx];
      if(((c>>>24)&255)<128) continue;
      r+=c&255; g+=(c>>>8)&255; b+=(c>>>16)&255; n++;
    }
    // próg 2 z 4: sprite zachowuje sylwetkę zamiast się rozpuszczać
    out[y*w2+x] = n<2 ? 0
      : (255<<24)|(((b/n)|0)<<16)|(((g/n)|0)<<8)|((r/n)|0);
  }
  return {w:w2,h:h2,data:out};
}

// zastępuje bezpośrednie wywołania shadeLevels(...) w rebuildArt/makeSprite
function buildTex(tex, n){
  const mips=[{w:tex.w,h:tex.h,data:tex.data}];
  while(mips.length<MIP_N){
    const m=mips[mips.length-1];
    if(m.w<=8||m.h<=8) break;
    mips.push(downsample(m.data,m.w,m.h));
  }
  tex.mip = mips.map(m => shadeLevels(m, n));
  tex.w=mips[0].w; tex.h=mips[0].h; tex.n=n;
  return tex;
}

Relief z mapy wysokości

Pakiet A
Cel
hmap z kroku A2 opisuje wypukłość każdego teksela. Gradient tej mapy, zestawiony ze stałym kierunkiem światła, daje wytłoczenie: kora wygląda na rzeźbioną, spoiny muru na wgłębione. W pakiecie A wypalamy je wprost w kolor tekstury — koszt runtime zero. W połączeniu z B1 ta sama tablica staje się reaktywna na światło.
Gdzie
Na końcu każdego generatora tekstur, przed grab(c) — a więc i przed mipmapami i przed shadeLevels.
Kod
// gradient mapy wysokości rzutowany na kierunek światła w płaszczyźnie
// tekstury; lx,ly ~ [-1,1], np. (-0.6,-0.8) = światło z góry-lewej
function relief(h,N,lx,ly){
  const out=new Uint8Array(N*N);
  for(let y=0;y<N;y++) for(let x=0;x<N;x++){
    const l=h[y*N+((x-1)&(N-1))], r=h[y*N+((x+1)&(N-1))];
    const u=h[((y-1)&(N-1))*N+x], d=h[((y+1)&(N-1))*N+x];
    const gx=(r-l)/255, gy=(d-u)/255;
    out[y*N+x] = clamp(128 - (gx*lx+gy*ly)*340, 0, 255)|0;
  }
  return out;
}

// 128 = neutralnie; siła 0.35 to dobry punkt startowy, 0.6 już przesadza
function applyRelief(px, ao, N, strength){
  for(let i=0;i<N*N;i++){
    const c=px[i];
    if(((c>>>24)&255)<128) continue;
    const k = 1 + (ao[i]/255 - 0.5)*2*strength;
    const r=(c&255)*k, g=((c>>>8)&255)*k, b=((c>>>16)&255)*k;
    px[i] = (255<<24)
          | ((b>255?255:b|0)<<16)
          | ((g>255?255:g|0)<<8)
          |  (r>255?255:r|0);
  }
}

// w texBark, tuż przed g.putImageData(im,0,0):
const ao = relief(hmap, N, -0.6, -0.8);
applyRelief(px, ao, N, 0.35);

let LM=null;                       // Float32Array, 3 kanały na komórkę, 1.0 = neutralnie
const LT=[0,0,0];                  // bufor roboczy, żeby nie alokować w pętli

function buildLightmap(B, seed){
  LM = new Float32Array(MW*MH*3);
  const rnd = mulberry32(seed);

  // 1. otwartość: ile nieba widzi komórka. Ważone jądro 7×7 to tania,
  //    ale zaskakująco przekonująca aproksymacja sky occlusion.
  const amb=new Float32Array(MW*MH);
  for(let y=0;y<MH;y++) for(let x=0;x<MW;x++){
    if(cell(x,y)!==0) continue;
    let open=0, tot=0;
    for(let dy=-3;dy<=3;dy++) for(let dx=-3;dx<=3;dx++){
      const w=1/(1+dx*dx+dy*dy);
      tot+=w; if(cell(x+dx,y+dy)===0) open+=w;
    }
    amb[y*MW+x]=open/tot;
  }

  // 2. wpisz ambient nieba — kolor bierzemy z biomu, siłę z otwartości
  const A=B.ambient, lo=B.ambientMin;
  for(let i=0;i<MW*MH;i++){
    const k = lo + (1-lo)*amb[i];
    LM[i*3]=A[0]*k; LM[i*3+1]=A[1]*k; LM[i*3+2]=A[2]*k;
  }

  // 3. źródła statyczne w losowych otwartych komórkach
  for(let i=0,put=0; i<400 && put<B.lights; i++){
    const x=1+(rnd()*(MW-2)|0), y=1+(rnd()*(MH-2)|0);
    if(cell(x,y)!==0) continue;
    if(amb[y*MW+x] > 0.75) continue;     // światła mają sens w gąszczu, nie na polanie
    addStaticLight(x+0.5, y+0.5, B.lightCol, B.lightRad);
    put++;
  }
}

function addStaticLight(lx,ly,col,rad){
  const x0=Math.max(0,(lx-rad)|0), x1=Math.min(MW-1,(lx+rad)|0);
  const y0=Math.max(0,(ly-rad)|0), y1=Math.min(MH-1,(ly+rad)|0);
  for(let cy=y0;cy<=y1;cy++) for(let cx=x0;cx<=x1;cx++){
    if(cell(cx,cy)!==0) continue;
    const d=Math.hypot(cx+0.5-lx, cy+0.5-ly);
    if(d>rad) continue;
    if(!losClear(lx,ly,cx+0.5,cy+0.5)) continue;   // istnieje już, linia 1674
    const f=(1-d/rad)*(1-d/rad);
    const i=(cy*MW+cx)*3;
    LM[i]+=col[0]*f; LM[i+1]+=col[1]*f; LM[i+2]+=col[2]*f;
  }
}

const DL=[];                        // światła dynamiczne, max MAXDL
const MAXDL=10;

function addLight(x,y,col,rad,life){
  if(DL.length>=MAXDL) DL.shift();  // najstarsze wypada; przy 10 nikt nie zauważy
  DL.push({x,y,c:col,rad,t:life,life});
}

// w update(dt), obok wygaszania cząstek:
for(let i=DL.length-1;i>=0;i--){
  DL[i].t-=dt;
  if(DL[i].t<=0) DL.splice(i,1);
}

// dopisz na końcu lightAt() z B2:
function addDynamic(px,py,out){
  for(let i=0;i<DL.length;i++){
    const L=DL[i];
    const dx=px-L.x, dy=py-L.y, d2=dx*dx+dy*dy;
    if(d2 > L.rad*L.rad) continue;
    const f=1-Math.sqrt(d2)/L.rad;
    const k=f*f*(L.t/L.life);       // kwadratowy spadek + wygasanie w czasie
    out[0]+=L.c[0]*k; out[1]+=L.c[1]*k; out[2]+=L.c[2]*k;
  }
}

To jest miejsce, w którym umiera budżet klatki, jeśli policzysz światło na piksel. Przy 10 światłach addDynamic to ok. 80 operacji. Kolor światła

Pakiet BMieszanka
Cel
Najlepszy przelicznik efektu do kosztu w całym dokumencie. Skoro B1 mnoży trzy kanały osobno, kolorowe światło nie kosztuje ani jednej dodatkowej operacji — to wyłącznie kwestia tego, jakie liczby wstawisz w ambient i lightCol. Wyłącznie dane: tabele w kroku B2 i kolory źródeł w B4. Zero zmian w pętlach.
Zasady
  • Ambient i źródła mają być w opozycji. Zimny ambient, ciepłe źródła — albo odwrotnie. Jeśli oba są w tej samej temperaturze, dostaniesz jaśniejszy obraz zamiast obrazu z rzeźbą.

  • Źródła mogą przekraczać 1.0. Błysk strzelby ma 1.60 na kanale czerwonym — clamp w B1 obetnie go do bieli w epicentrum, co jest dokładnie tym, czego chcesz od lufy.
  • Nie przebarwiaj znajdziek. W buildSprites celowo pomijają treeTint; światło przebarwi je mimo to. Jeśli apteczka przestaje być czytelna w GRZYBNI, daj sprite'om znajdziek mnożnik ściągnięty ku szarości: lr = 256 + (lr-256)*0.35.
Koszt
Kod +1 KB, w całości dane. CPU zero. RAM zero.
Sprawdzian
Przejdź wszystkie pięć biomów i zrób zrzut z tego samego miejsca. Każdy powinien mieć rozpoznawalną dominantę barwną — jeśli dwa wyglądają podobnie, ambienty są za blisko siebie.

Pakiet BMieszanka
Cel

B10

Cień kontaktowy pod sprite'ami

Pakiet BMieszanka
Cel
Sprite'y w raycasterze zawsze sprawiają wrażenie lewitujących, bo nic nie łączy ich z podłożem. Ciemna elipsa u podstawy, rysowana tuż przed sprite'em, likwiduje to jednym ruchem. Przy znajdźkach, które mają zoff i unoszą się, cień zostaje na ziemi — i dopiero wtedy widać, że one faktycznie lewitują.
Gdzie
drawSprite (2067), na początku — po wyliczeniu scr, floorY i unit, przed pętlami rysującymi teksturę.
Kod
function contactShadow(scr, floorY, unit, wUnits, tY, str){
  const rw = (unit*wUnits*0.55)|0;
  const rh = Math.max(1, (rw*0.34)|0);
  if(rw<1) return;
  const fy=floorY|0;
  const y0=Math.max(0,fy-rh), y1=Math.min(H-1,fy+rh);
  const x0=Math.max(0,scr-rw), x1=Math.min(W-1,scr+rw);
  for(let y=y0;y<=y1;y++){
    const dy=(y-fy)/rh, dy2=dy*dy;
    const row=y*W;
    for(let x=x0;x<=x1;x++){
      if(zbuf[x] <= tY) continue;          // ściana bliżej — cień nie wychodzi
      const dx=(x-scr)/rw;
      const d=dx*dx+dy2;
      if(d>=1) continue;
      const k = 256 - (str*(1-d)*256)|0;   // mnożnik < 256 = przyciemnienie
      const c = buf[row+x];
      buf[row+x] = 0xFF000000
        | ((((c>>>16&255)*k)>>8)<<16)
        | ((((c>>>8 &255)*k)>>8)<<8)
        |  (((c     &255)*k)>>8);
    }
  }
}

// w drawSprite, po wyliczeniu floorY i przed rysowaniem tekstury.
// UWAGA: cień idzie na wysokość GRUNTU, więc bez odejmowania zoff —
// dzięki temu unosząca się znajdźka rzuca cień tam, gdzie leży ziemia.
const groundY = H/2 + camShift + unit*0.5;
contactShadow(scr, groundY, unit, s.wUnits, tY, 0.5);
Kolejność
Nic nie trzeba przestawiać w render(). Lista sprite'ów jest już posortowana od najdalszego do najbliższego, a cień rysuje się wewnątrz tej samej iteracji co jego sprite — więc cień bliższego drzewa pada na dalsze, już narysowane, co jest zachowaniem prawidłowym. Dwa osobne przebiegi (najpierw wszystkie cienie, potem wszystkie sprite'y) tę własność by zepsuły.
Pułapki
Test zbuf[x] <= tY jest obowiązkowy. Bez niego cień drzewa stojącego za ścianą przecieka na tę ścianę. To ten sam warunek, którego używa pętla rysująca sprite tuż niżej — po prostu wykonaj go także tutaj.

Siła 0.5 przy dużym drzewie na wprost potrafi przykryć pół ekranu ciemną plamą. Jeśli to razi, zwiąż siłę z rozmiarem: str = 0.5 * clamp(3/tY, 0.3, 1) — daleki cień pozostaje mocny, bliski się wycofuje.

Cień nie ma testu głębi w pionie, więc położony na stromym zboczu wyglądałby źle. W tej grze podłoga jest płaska, więc problem nie występuje — ale nie przenoś tego kodu do projektu z różnicami wysokości bez zmian.

Koszt
Kod +1,5 KB. CPU: elipsa ma zwykle poniżej 2 000 pikseli, przy 40 sprite'ach ok. 80 000 przetworzeń — +8–12% na renderze, przy czym większość przypada na bliskie drzewa, kiedy i tak mało co jest rysowane.
Sprawdzian
Podejdź do leżącej znajdźki. Cień ma zostać na ziemi i nie podskakiwać razem z unoszącym się przedmiotem — jeśli podskakuje, gdzieś przeszło zoff do groundY.
